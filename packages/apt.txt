# Benodigde pakketten uit de Ubuntu-repo's (universe ingeschakeld)
# Installeren:
#   sudo apt update
#   sudo apt install -y qtile rofi kitty picom dunst feh imagemagick \
#     x11-xserver-utils fonts-font-awesome python3 python3-pip pipx \
#     git build-essential python3-setuptools libnotify-bin \
#     pulseaudio-utils bluez blueman pavucontrol libglib2.0-bin procps curl
#
# pywal16 staat niet in de Ubuntu-repo's; installeer het met pipx:
#   pipx install pywal16
# Zorg dat ~/.local/bin in je PATH staat (pipx ensurepath; daarna opnieuw inloggen).

qtile               # window manager (X11)
rofi                # applicatielauncher en wallpaperkiezer
kitty               # terminal (kleuren via pywal)
picom               # compositor (schaduwen, transparantie, afgeronde hoeken, blur)
dunst               # notificatiedaemon (thema volgt pywal)
feh                 # wallpaper-viewer (optioneel; qtile tekent de achtergrond zelf)
imagemagick         # kleur- en thumbnail-generatie (pywal + rofi-galerij)
x11-xserver-utils   # xrdb, zodat X-apps de pywal-kleuren gebruiken
fonts-font-awesome  # pictogrammen in de qtile-balk (o.a. power-knop)
python3             # qtile-config en de hulpscripts
python3-pip         # pipx/pywal16
pipx                # pywal16 in een geïsoleerde omgeving
git                 # repo clonen
build-essential     # compiler e.d. voor Python-pakketten
python3-setuptools  # hulpmiddelen voor Python-pakketten
libnotify-bin       # notify-send (wallpaperkiezer)
pulseaudio-utils    # pactl (Bluetooth-audio; werkt ook met PipeWire)
bluez               # bluetoothctl
blueman             # blueman-applet in de systray
pavucontrol         # geluidsinstellingen vanuit de qtile-balk
libglib2.0-bin      # gsettings (GTK prefer-dark)
procps              # pgrep (systemd ExecCondition)
curl                # wallpapers ophalen
